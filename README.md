```sql
    customer_id,
    COUNT(order_id) AS total_orders,
    SUM(quantity * unit_price) AS revenue
FROM orders
GROUP BY customer_id
ORDER BY revenue DESC
LIMIT 10;
```

```python
import pandas as pd

df = pd.read_csv("data.csv")

df = df.drop_duplicates()

df["revenue"] = (
    df["quantity"] *
    df["unit_price"]
)

result = (
    df.groupby("category")
      ["revenue"]
      .sum()
      .sort_values(ascending=False)
)

print(result)
```
